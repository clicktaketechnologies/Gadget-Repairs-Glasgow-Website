   147→function loadFromStorage(): Booking[] | null {
   148→  if (typeof window === "undefined") return null;
   149→  try {
   150→    // Read admin bookings
   151→    const raw = localStorage.getItem(STORAGE_KEY);
   152→    const adminBookings = raw ? JSON.parse(raw) : [];
   153→
   154→    // Also read public bookings (submitted via the website booking form)
   155→    const publicRaw = localStorage.getItem(PUBLIC_KEY);
   156→    const publicBookings = publicRaw ? JSON.parse(publicRaw) : [];
   157→
   158→    // Merge: public bookings first (most recent), then admin bookings (dedup by id)
   159→    const adminIds = new Set(adminBookings.map((b: any) => b.id));
   160→    const merged = [
   161→      ...publicBookings.filter((b: any) => !adminIds.has(b.id)),
   162→      ...adminBookings,
   163→    ];
   164→
   165→    if (Array.isArray(merged) && merged.length > 0) return merged;
   166→    return null;
   167→  } catch {
   168→    return null;
   169→  }
   170→}
   171→
   172→function saveToStorage(bookings: Booking[]) {
   173→  if (typeof window === "undefined") return;
   174→  try {
   175→    // Save admin bookings (exclude public ones by id prefix)
   176→    const adminOnly = bookings.filter((b) => !b.id.startsWith("pub-"));
   177→    localStorage.setItem(STORAGE_KEY, JSON.stringify(adminOnly));
   178→
   179→    // Only update public bookings if there are any in the current list
   180→    // (don't remove the public key — public bookings might exist outside this list)
   181→    const publicBookings = bookings.filter((b) => b.id.startsWith("pub-"));
   182→    if (publicBookings.length > 0) {
   183→      localStorage.setItem(PUBLIC_KEY, JSON.stringify(publicBookings));
   184→    }
   185→    // NEVER removeItem(PUBLIC_KEY) — it would delete customer submissions
   186→  } catch {}
   187→}
   188→
   189→function loadTrashFromStorage(): Booking[] {
   190→  if (typeof window === "undefined") return [];
   191→  try {
   192→    const raw = localStorage.getItem(TRASH_KEY);
   193→    return raw ? JSON.parse(raw) : [];
   194→  } catch {
   195→    return [];
   196→  }
   197→}

   200→  if (typeof window === "undefined") return;
   201→  try {
   202→    localStorage.setItem(TRASH_KEY, JSON.stringify(trash));
   203→  } catch {}
   204→}

   206→function saveSentEmail(email: SentEmail) {
   207→  if (typeof window === "undefined") return;
   208→  try {
   209→    const raw = localStorage.getItem(SENT_EMAILS_KEY);
   210→    const existing: SentEmail[] = raw ? JSON.parse(raw) : [];
   211→    existing.unshift(email);
   212→    localStorage.setItem(SENT_EMAILS_KEY, JSON.stringify(existing));
   213→  } catch {}
   214→}
