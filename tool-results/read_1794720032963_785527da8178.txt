   147→   147→function loadFromStorage(): Booking[] | null {
   148→   148→  if (typeof window === "undefined") return null;
   149→   149→  try {
   150→   150→    // Read admin bookings
   151→   151→    const raw = localStorage.getItem(STORAGE_KEY);
   152→   152→    const adminBookings = raw ? JSON.parse(raw) : [];
   153→   153→
   154→   154→    // Also read public bookings (submitted via the website booking form)
   155→   155→    const publicRaw = localStorage.getItem(PUBLIC_KEY);
   156→   156→    const publicBookings = publicRaw ? JSON.parse(publicRaw) : [];
   157→   157→
   158→   158→    // Merge: public bookings first (most recent), then admin bookings (dedup by id)
   159→   159→    const adminIds = new Set(adminBookings.map((b: any) => b.id));
   160→   160→    const merged = [
   161→   161→      ...publicBookings.filter((b: any) => !adminIds.has(b.id)),
   162→   162→      ...adminBookings,
   163→   163→    ];
   164→   164→
   165→   165→    if (Array.isArray(merged) && merged.length > 0) return merged;
   166→   166→    return null;
   167→   167→  } catch {
   168→   168→    return null;
   169→   169→  }
   170→   170→}
   171→   171→
   172→   172→function saveToStorage(bookings: Booking[]) {
   173→   173→  if (typeof window === "undefined") return;
   174→   174→  try {
   175→   175→    // Save admin bookings (exclude public ones by id prefix)
   176→   176→    const adminOnly = bookings.filter((b) => !b.id.startsWith("pub-"));
   177→   177→    localStorage.setItem(STORAGE_KEY, JSON.stringify(adminOnly));
   178→   178→
   179→   179→    // Only update public bookings if there are any in the current list
   180→   180→    // (don't remove the public key — public bookings might exist outside this list)
   181→   181→    const publicBookings = bookings.filter((b) => b.id.startsWith("pub-"));
   182→   182→    if (publicBookings.length > 0) {
   183→   183→      localStorage.setItem(PUBLIC_KEY, JSON.stringify(publicBookings));
   184→   184→    }
   185→   185→    // NEVER removeItem(PUBLIC_KEY) — it would delete customer submissions
   186→   186→  } catch {}
   187→   187→}
