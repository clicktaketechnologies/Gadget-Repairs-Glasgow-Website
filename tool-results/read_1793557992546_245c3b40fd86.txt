   106→   106→function loadFromStorage(): Booking[] | null {
   107→   107→  if (typeof window === "undefined") return null;
   108→   108→  try {
   109→   109→    // Read admin bookings
   110→   110→    const raw = localStorage.getItem(STORAGE_KEY);
   111→   111→    const adminBookings = raw ? JSON.parse(raw) : [];
   112→   112→
   113→   113→    // Also read public bookings (submitted via the website booking form)
   114→   114→    const publicRaw = localStorage.getItem(PUBLIC_KEY);
   115→   115→    const publicBookings = publicRaw ? JSON.parse(publicRaw) : [];
   116→   116→
   117→   117→    // Merge: public bookings first (most recent), then admin bookings (dedup by id)
   118→   118→    const adminIds = new Set(adminBookings.map((b: any) => b.id));
   119→   119→    const merged = [
   120→   120→      ...publicBookings.filter((b: any) => !adminIds.has(b.id)),
   121→   121→      ...adminBookings,
   122→   122→    ];
   123→   123→
   124→   124→    if (Array.isArray(merged) && merged.length > 0) return merged;
   125→   125→    return null;
   126→   126→  } catch {
   127→   127→    return null;
   128→   128→  }
   129→   129→}
   130→   130→
   131→   131→function saveToStorage(bookings: Booking[]) {
   132→   132→  if (typeof window === "undefined") return;
   133→   133→  try {
   134→   134→    // Save admin bookings (exclude public ones by id prefix)
   135→   135→    const adminOnly = bookings.filter((b) => !b.id.startsWith("local-") && !b.id.startsWith("public-"));
   136→   136→    localStorage.setItem(STORAGE_KEY, JSON.stringify(adminOnly));
   137→   137→
   138→   138→    // Also update public bookings if any public bookings were modified
   139→   139→    const publicBookings = bookings.filter((b) => b.id.startsWith("local-") || b.id.startsWith("public-"));
   140→   140→    if (publicBookings.length > 0) {
   141→   141→      localStorage.setItem(PUBLIC_KEY, JSON.stringify(publicBookings));
   142→   142→    }
   143→   143→  } catch {}
   144→   144→}
