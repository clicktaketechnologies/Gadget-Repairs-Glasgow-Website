   106→function loadFromStorage(): Booking[] | null {
   107→  if (typeof window === "undefined") return null;
   108→  try {
   109→    // Read admin bookings
   110→    const raw = localStorage.getItem(STORAGE_KEY);
   111→    const adminBookings = raw ? JSON.parse(raw) : [];
   112→
   113→    // Also read public bookings (submitted via the website booking form)
   114→    const publicRaw = localStorage.getItem(PUBLIC_KEY);
   115→    const publicBookings = publicRaw ? JSON.parse(publicRaw) : [];
   116→
   117→    // Merge: public bookings first (most recent), then admin bookings (dedup by id)
   118→    const adminIds = new Set(adminBookings.map((b: any) => b.id));
   119→    const merged = [
   120→      ...publicBookings.filter((b: any) => !adminIds.has(b.id)),
   121→      ...adminBookings,
   122→    ];
   123→
   124→    if (Array.isArray(merged) && merged.length > 0) return merged;
   125→    return null;
   126→  } catch {
   127→    return null;
   128→  }
   129→}
   130→
   131→function saveToStorage(bookings: Booking[]) {
   132→  if (typeof window === "undefined") return;
   133→  try {
   134→    // Save admin bookings (exclude public ones by id prefix)
   135→    const adminOnly = bookings.filter((b) => !b.id.startsWith("local-") && !b.id.startsWith("public-"));
   136→    localStorage.setItem(STORAGE_KEY, JSON.stringify(adminOnly));
   137→
   138→    // Also update public bookings if any public bookings were modified
   139→    const publicBookings = bookings.filter((b) => b.id.startsWith("local-") || b.id.startsWith("public-"));
   140→    if (publicBookings.length > 0) {
   141→      localStorage.setItem(PUBLIC_KEY, JSON.stringify(publicBookings));
   142→    }
   143→  } catch {}
   144→}

   213→  async function saveBooking() {
   214→    if (!selected) return;
   215→    setSaving(true);
   216→    const now = new Date().toISOString();
   217→    try {
   218→      const data = await adminJson<{ booking: Booking }>(
   219→        `/api/admin/bookings/${selected.id}`,
   220→        {
   221→          method: "PATCH",
   222→          body: JSON.stringify({
   223→            status: editStatus,
   224→            technicianNotes: editNotes || null,
   225→            quotedPrice: editQuoted || null,
   226→            finalPrice: editFinal || null,
   227→          }),
   228→        }
   229→      );
   230→      setBookings((prev) =>
   231→        prev.map((b) => (b.id === data.booking.id ? data.booking : b))
   232→      );
   233→      setSelected(data.booking);
   234→      toast.success("Booking updated");
   235→    } catch {
   236→      // API failed — save locally so data persists across refreshes
   237→      const updated: Booking = {
   238→        ...selected,
   239→        status: editStatus,
   240→        technicianNotes: editNotes || null,
   241→        quotedPrice: editQuoted || null,
   242→        finalPrice: editFinal || null,
   243→        updatedAt: now,
   244→      };
   245→      setBookings((prev) => {
   246→        const next = prev.map((b) => (b.id === updated.id ? updated : b));
   247→        saveToStorage(next);
   248→        return next;
   249→      });
   250→      setSelected(updated);
   251→      toast.success("Booking updated (saved locally)");
   252→    } finally {
   253→      setSaving(false);
   254→    }
   255→  }

   261→  async function confirmDelete() {
   262→    if (!selected) return;

   266→    setBookings((prev) => {
   267→      const next = prev.filter((b) => b.id !== idToDelete);
   268→      saveToStorage(next);
   269→      return next;

   273→        method: "DELETE",
   274→      });

   276→    } catch {
   277→      // API failed — booking already removed locally, that's fine
   278→      toast.success("Booking deleted (saved locally)");
   279→    }
   280→    setDeleteOpen(false);
   281→    setDetailOpen(false);
   282→    setSelected(null);
   283→    setDeleting(false);
   284→  }
