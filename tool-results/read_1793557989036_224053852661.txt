    83→function loadFromStorage(): BlogPost[] | null {
    84→  if (typeof window === "undefined") return null;
    85→  try {
    86→    const raw = localStorage.getItem(STORAGE_KEY);
    87→    if (!raw) return null;
    88→    const parsed = JSON.parse(raw);
    89→    if (Array.isArray(parsed) && parsed.length > 0) return parsed;
    90→    return null;
    91→  } catch {
    92→    return null;
    93→  }
    94→}

    96→function saveToStorage(posts: BlogPost[]) {
    97→  if (typeof window === "undefined") return;
    98→  try {
    99→    localStorage.setItem(STORAGE_KEY, JSON.stringify(posts));
   100→  } catch {}
   101→}

   257→          ...form,
   258→          slug,
   259→          updatedAt: now,

   264→          saveToStorage(next);
   265→          return next;
   266→        });
   267→        toast.success("Post updated (saved locally)");

   270→          id: `local-${Date.now()}`,

   284→        };

   288→          return next;
   289→        });

   291→      }
   292→      setCreating(false);
   293→    } finally {
   294→      setSaving(false);

   298→  async function patchField(p: BlogPost, field: "published" | "featured", value: boolean) {
   299→    // Optimistic update — save to state + localStorage immediately
   300→    setPosts((prev) => {
   301→      const next = prev.map((x) => (x.id === p.id ? { ...x, [field]: value, updatedAt: new Date().toISOString() } : x));
   302→      saveToStorage(next);
   303→      return next;
   304→    });
   305→    try {
   306→      await adminJson(`/api/admin/blog/${p.id}`, {
   307→        method: "PATCH",
   308→        body: JSON.stringify({ [field]: value }),
   309→      });
   310→      toast.success(`${p.title}: ${field} ${value ? "on" : "off"}`);
   311→    } catch {
   312→      // API failed — keep the optimistic update (don't revert!)
   313→      toast.success(`${p.title}: ${field} ${value ? "on" : "off"} (saved locally)`);
   314→    }
   315→  }
